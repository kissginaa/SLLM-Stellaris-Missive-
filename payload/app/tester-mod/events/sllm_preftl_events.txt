# SLLM 원시문명 관측(2026-11-03): 처음 접촉한 원시문명을 앱에 알린다(SLLM:PFTL_FOUND) → 앱이 그때 한 번 관측 명령(sllm_probe_preftl)을 보낸다.
# 그 뒤로는 자동 관측 없음(플레이어가 화면을 열 때만). 게임 효과 없음(표식 sllm_pf_known 만). 앱: src/features/preftl-app.ts.
namespace = sllm_preftl

# 조사선이 행성을 조사함(바닐라 on_planet_surveyed: Root = 행성, From = 조사한 나라). 원시문명 행성이고 처음이면 알린다.
planet_event = {
	id = sllm_preftl.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = { is_primitive = yes NOT = { has_country_flag = sllm_pf_known } }
		from = { is_ai = no }
	}
	immediate = {
		owner = { set_country_flag = sllm_pf_known }
		from = { log = "SLLM:PFTL_FOUND/;/sid=\\[This.sllm_save_id]/;/how=survey/;/name=\\[Root.Owner.GetName]" }
	}
}

# 관측소를 지음(바닐라 on_building_observation_station: This = 건설선, From = 행성). 플레이어가 지었으면 알린다(이미 아는 문명이어도 임무가 바뀌므로).
ship_event = {
	id = sllm_preftl.2
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		owner = { is_ai = no }
		from = { exists = owner owner = { is_primitive = yes } }
	}
	immediate = {
		from = { owner = { set_country_flag = sllm_pf_known } }
		owner = { log = "SLLM:PFTL_FOUND/;/sid=\\[This.sllm_save_id]/;/how=post/;/name=\\[From.Owner.GetName]" }
	}
}

# sllm_inf.11 식민지가 폭격으로 말살될 때(on_colony_destroyed_by_bombardment, from = 폭격한 나라): +10

namespace = sllm_inf

country_event = {
	id = sllm_inf.1
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_variable_set = sllm_inf check_variable = { which = sllm_inf value > 0 } }
	immediate = {
		if = {
			limit = { is_at_war = no }
			subtract_variable = { which = sllm_inf value = 0.5 }
			if = { limit = { check_variable = { which = sllm_inf value < 0 } } set_variable = { which = sllm_inf value = 0 } }
			sllm_inf_refresh = yes
		}
	}
}

# 3단계(2026-11-04): 숨은 악명이 드러날지(해마다, 플레이어만 쌓임). 10% + 숨은 악명 10/25/40 이상이면 +10/+15/+20. 안 드러나면 증거가 흐려져 0.85배.
country_event = {
	id = sllm_inf.5
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_variable_set = sllm_inf_hid check_variable = { which = sllm_inf_hid value > 0.5 } }
	immediate = {
		random_list = {
			10 = {
				modifier = { add = 10 check_variable = { which = sllm_inf_hid value >= 10 } }
				modifier = { add = 15 check_variable = { which = sllm_inf_hid value >= 25 } }
				modifier = { add = 20 check_variable = { which = sllm_inf_hid value >= 40 } }
				sllm_inf_expose = yes
			}
			90 = {
				multiply_variable = { which = sllm_inf_hid value = 0.85 }
			}
		}
	}
}

# 3단계: 은폐 공작(앱 요청, root = 플레이어). 영향력 75·에너지 300, 2년에 한 번. 공개 악명 최대 15를 숨은 악명으로 옮긴다.
# 결과 st: 1 됨 · 2 자원 부족 · 3 아직 때가 아님 · 4 옮길 악명 없음
country_event = {
	id = sllm_inf.6
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_variable = { which = sllm_inf_st value = 0 }
		set_variable = { which = sllm_inf_mv value = 0 }
		if = { limit = { NOT = { is_variable_set = sllm_inf } } set_variable = { which = sllm_inf value = 0 } }
		if = { limit = { NOT = { is_variable_set = sllm_inf_hid } } set_variable = { which = sllm_inf_hid value = 0 } }
		if = { limit = { has_country_flag = sllm_inf_cover_cd } set_variable = { which = sllm_inf_st value = 3 } }
		else_if = { limit = { check_variable = { which = sllm_inf value < 1 } } set_variable = { which = sllm_inf_st value = 4 } }
		else_if = {
			limit = { OR = { NOT = { resource_stockpile_compare = { resource = influence value >= 75 } } NOT = { resource_stockpile_compare = { resource = energy value >= 300 } } } }
			set_variable = { which = sllm_inf_st value = 2 }
		}
		else = {
			add_resource = { influence = -75 energy = -300 }
			set_variable = { which = sllm_inf_mv value = sllm_inf }
			if = { limit = { check_variable = { which = sllm_inf_mv value > 15 } } set_variable = { which = sllm_inf_mv value = 15 } }
			subtract_variable = { which = sllm_inf value = sllm_inf_mv }
			change_variable = { which = sllm_inf_hid value = sllm_inf_mv }
			set_timed_country_flag = { flag = sllm_inf_cover_cd days = 720 }
			sllm_inf_refresh = yes
			set_variable = { which = sllm_inf_st value = 1 }
		}
		log = "SLLM:INFCOVER/;/sid=\\[This.sllm_save_id]/;/req=\\[This.sllm_inf_req]/;/st=\\[This.sllm_inf_st]/;/mv=\\[This.sllm_inf_mv]/;/inf=\\[This.sllm_inf]/;/hid=\\[This.sllm_inf_hid]"
	}
}

# SLLM Probe - 집무실 연락처: 내 지도자 명단(2026-10-18). 플레이어 국가만, 매달 한 번. 게임 규칙은 바꾸지 않는다.
# 로그: 지도자마다 SLLM:LEADER(lid·name·cls·lv·ruler·council·tr·cpos·asg) → tr=1 이면 SLLM:LTRAIT(lid·t) 여러 줄(scripted_effects/sllm_probe_leader_traits.txt, 도구로 생성) → 함대가 있으면 SLLM:LEADERAT(lid·fleet) → 끝에 SLLM:LEADER_END(date).
# 지도자 번호 sllm_lid 는 함대 사령부(sllm_fleet)와 같은 번호를 쓴다(국가 변수 sllm_lid_next 에서 나눠 줌). 이름이 바뀌어도 번호로 같은 사람을 잇는다.
# cls: commander(지휘관)·official(관료·총독)·scientist(과학자). lv: 기술 수준(has_skill 로 1~10).
namespace = sllm_people

# 월간 명단 미루기(2026-10-24 렉 줄이기): 매달 1일에 다른 조회와 겹치지 않게 6일 뒤에 sllm_people.1 을 부른다.
# 「슈뢰딩거」(2026-11-03 사용자): 이제 매달 명단을 다시 읽지 않는다. 지난달에 지도자가 바뀐 일(고용·해임·사망·승급·배치·통치자 교체 → 표식 sllm_roster_dirty)이 있었거나,
# 특성을 다시 읽어야 할 지도자(sllm_tr_m ≥ 12: 승급·SLLM 특성 명령)가 있을 때만 읽는다. 그 밖에는 1년에 한 번(sllm_people.8)과 앱이 집무실 연락처를 열 때(앱이 sllm_people.1 을 직접 부름).
country_event = {
	id = sllm_people.9
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_ai = no }
	immediate = {
		if = {
			limit = { OR = { has_country_flag = sllm_roster_dirty any_owned_leader = { is_variable_set = sllm_tr_m check_variable = { which = sllm_tr_m value >= 12 } } } }
			remove_country_flag = sllm_roster_dirty
			country_event = { id = sllm_people.1 days = 6 }
		}
	}
}

# 지도자 승급(on_leader_level_up, This = 국가, From = 지도자, 2026-10-24): 특성이 새로 붙었을 수 있으니 다음 명단에서 특성을 읽게 한다.

country_event = {
	id = sllm_people.1
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_ai = no }
	immediate = {
		# 총독 표식(2026-10-23 수정): 지도자 scope 의 sector 는 「지금 있는 구역」이지 총독 임명이 아니다(게임 script_documentation/scopes.log).
		# 그래서 행성·구역에서 거꾸로 총독을 찾아 sllm_isgov=1 로 표시하고, asg 는 함대 또는 이 표식으로만 판단한다.
		every_owned_leader = { set_variable = { which = sllm_isgov value = 0 } }
		# 구역(sector) 범위에서는 leader 로 넘어갈 수 없다(2026-10-27 테스터 error.log, 4.5 scopes.log) → 식민지 쪽 leader 로만 찾는다.
		every_owned_colony = { limit = { exists = leader } leader = { set_variable = { which = sllm_isgov value = 1 } } }
		every_owned_leader = {
			if = { limit = { NOT = { is_variable_set = sllm_lid } }
				if = { limit = { root = { NOT = { is_variable_set = sllm_lid_next } } } root = { set_variable = { which = sllm_lid_next value = 1 } } }
				set_variable = { which = sllm_lid value = root.sllm_lid_next }
				root = { change_variable = { which = sllm_lid_next value = 1 } }
			}
			set_variable = { which = sllm_lv value = 0 }
			if = { limit = { has_base_skill >= 10 } set_variable = { which = sllm_lv value = 10 } }
			else_if = { limit = { has_base_skill >= 9 } set_variable = { which = sllm_lv value = 9 } }
			else_if = { limit = { has_base_skill >= 8 } set_variable = { which = sllm_lv value = 8 } }
			else_if = { limit = { has_base_skill >= 7 } set_variable = { which = sllm_lv value = 7 } }
			else_if = { limit = { has_base_skill >= 6 } set_variable = { which = sllm_lv value = 6 } }
			else_if = { limit = { has_base_skill >= 5 } set_variable = { which = sllm_lv value = 5 } }
			else_if = { limit = { has_base_skill >= 4 } set_variable = { which = sllm_lv value = 4 } }
			else_if = { limit = { has_base_skill >= 3 } set_variable = { which = sllm_lv value = 3 } }
			else_if = { limit = { has_base_skill >= 2 } set_variable = { which = sllm_lv value = 2 } }
			else_if = { limit = { has_base_skill >= 1 } set_variable = { which = sllm_lv value = 1 } }
			# 특성(2026-10-18 → 2026-10-24 렉 줄이기): 처음 보는 지도자는 바로, 승급했거나 SLLM 이 특성을 붙이거나 뗐으면 다음 명단에서, 그 밖에는 1년(12달)마다 한 번만 읽는다(759개 검사라 자주 하지 않는다).
			# 승급(sllm_people.6)과 앱의 특성 명령은 sllm_tr_m 을 12 로 올려 다음 명단에서 바로 읽게 한다. tr=1 이면 이번 달 특성 목록이 함께 온다.
			set_variable = { which = sllm_tr value = 0 }
			if = { limit = { OR = { NOT = { is_variable_set = sllm_tr_m } check_variable = { which = sllm_tr_m value >= 12 } root = { has_country_flag = sllm_tr_all } } }
				set_variable = { which = sllm_tr value = 1 } set_variable = { which = sllm_tr_m value = 1 }
			}
			else = { change_variable = { which = sllm_tr_m value = 1 } }
			set_variable = { which = sllm_rl value = 0 }
			if = { limit = { is_ruler = yes } set_variable = { which = sllm_rl value = 1 } }
			set_variable = { which = sllm_cn value = 0 }
			if = { limit = { is_councilor = yes } set_variable = { which = sllm_cn value = 1 } }
			# 평의회 직책(2026-10-19, 내각 회의): 평의원이면 직책 번호 sllm_cp(scripted_effects/sllm_probe_council_pos.txt), 아니면 0.
			# 내각 회의(2026-10-19): 함대·과학선을 맡았거나(fleet) 총독이면(위 sllm_isgov 표식) asg=1 — 회의에는 홀로그램으로 참석, 아니면 직접 참석.
			set_variable = { which = sllm_asg value = 0 }
			if = { limit = { OR = { exists = fleet check_variable = { which = sllm_isgov value = 1 } } } set_variable = { which = sllm_asg value = 1 } }
			set_variable = { which = sllm_cp value = 0 }
			if = { limit = { is_councilor = yes is_ruler = no } sllm_probe_council_pos = yes }
			# 나이·수명·복무 연수·불멸(2026-10-23 은퇴 생활, 4.5.1 triggers.log 의 leader scope 트리거). 은퇴할 때의 마지막 값을 앱이 보관한다.
			set_variable = { which = sllm_age value = -1 } set_variable = { which = sllm_life value = -1 } set_variable = { which = sllm_svc value = -1 }
			export_trigger_value_to_variable = { trigger = leader_age variable = sllm_age }
			export_trigger_value_to_variable = { trigger = leader_lifespan variable = sllm_life }
			export_trigger_value_to_variable = { trigger = leader_years_of_service variable = sllm_svc }
			set_variable = { which = sllm_imm value = 0 }
			if = { limit = { is_immortal = yes } set_variable = { which = sllm_imm value = 1 } }
			if = { limit = { leader_class = commander } log = "SLLM:LEADER/;/lid=\\[This.sllm_lid]/;/cls=commander/;/lv=\\[This.sllm_lv]/;/ruler=\\[This.sllm_rl]/;/council=\\[This.sllm_cn]/;/tr=\\[This.sllm_tr]/;/cpos=\\[This.sllm_cp]/;/asg=\\[This.sllm_asg]/;/name=\\[This.GetName]/;/age=\\[This.sllm_age]/;/life=\\[This.sllm_life]/;/svc=\\[This.sllm_svc]/;/imm=\\[This.sllm_imm]" }
			else_if = { limit = { leader_class = scientist } log = "SLLM:LEADER/;/lid=\\[This.sllm_lid]/;/cls=scientist/;/lv=\\[This.sllm_lv]/;/ruler=\\[This.sllm_rl]/;/council=\\[This.sllm_cn]/;/tr=\\[This.sllm_tr]/;/cpos=\\[This.sllm_cp]/;/asg=\\[This.sllm_asg]/;/name=\\[This.GetName]/;/age=\\[This.sllm_age]/;/life=\\[This.sllm_life]/;/svc=\\[This.sllm_svc]/;/imm=\\[This.sllm_imm]" }
			else = { log = "SLLM:LEADER/;/lid=\\[This.sllm_lid]/;/cls=official/;/lv=\\[This.sllm_lv]/;/ruler=\\[This.sllm_rl]/;/council=\\[This.sllm_cn]/;/tr=\\[This.sllm_tr]/;/cpos=\\[This.sllm_cp]/;/asg=\\[This.sllm_asg]/;/name=\\[This.GetName]/;/age=\\[This.sllm_age]/;/life=\\[This.sllm_life]/;/svc=\\[This.sllm_svc]/;/imm=\\[This.sllm_imm]" }
			if = { limit = { check_variable = { which = sllm_tr value = 1 } } sllm_probe_leader_traits = yes }
			if = { limit = { exists = fleet }
				root = { set_variable = { which = sllm_pl_cur value = prev.sllm_lid } }
				fleet = { log = "SLLM:LEADERAT/;/lid=\\[Root.sllm_pl_cur]/;/fleet=\\[This.GetName]" }
			}
			# 변수 줄이기(2026-10-24): 기록에만 쓰는 임시 값은 지운다(세이브에 지도자마다 쌓이지 않게). sllm_lid·sllm_tr_m 은 남긴다.
			clear_variable = sllm_lv clear_variable = sllm_tr clear_variable = sllm_rl clear_variable = sllm_cn clear_variable = sllm_asg clear_variable = sllm_cp
			clear_variable = sllm_age clear_variable = sllm_life clear_variable = sllm_svc clear_variable = sllm_imm clear_variable = sllm_isgov
		}
		clear_variable = sllm_pl_cur
		remove_country_flag = sllm_tr_all
		log = "SLLM:LEADER_END/;/date=\\[GetDate]"
	}
}

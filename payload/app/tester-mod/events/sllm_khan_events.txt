# SLLM 그레이트 칸 특별 대우(다음 업데이트, 2026-10-26 외부 건의 ④). 앱: src/khan.ts · src/features/khan-app.ts.
# 사용자 확정: 「그 부족과 전투한 적」이 없고 「호감 +50 이상 또는 거래 이력」이면 → 그레이트 칸의 사자 통신·특별 대화, 선물(합금 월수입 6개월치), 우대 태수국(칸의 총애: 기본 자원 생산 +15%, 그레이트 칸의 태수국인 동안).
# 바닐라 태수국 조건(preset_satrapy, 기본 자원 30%)은 바꾸지 않는다 — 다른 바닐라 약탈자 이벤트가 그 조건 이름을 검사하기 때문.
namespace = sllm_khan

# 우주 전투가 끝남(이김·짐 모두): 상대가 약탈자 부족(잠든·습격대·각성)이면 플레이어에게 그 부족과 싸웠다는 표식. 바닐라 부족 표식 marauder_1~3.
# 범위(바닐라 00_on_actions 주석): This = 함대 1 소유국, From = 함대 2 소유국. 어느 쪽이 플레이어여도 기록한다.
country_event = {
	id = sllm_khan.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { is_ai = no from = { OR = { is_country_type = dormant_marauders is_country_type = marauder_raiders is_country_type = awakened_marauders } } }
			if = { limit = { from = { has_country_flag = marauder_1 } } set_country_flag = sllm_fought_marauder_1 }
			if = { limit = { from = { has_country_flag = marauder_2 } } set_country_flag = sllm_fought_marauder_2 }
			if = { limit = { from = { has_country_flag = marauder_3 } } set_country_flag = sllm_fought_marauder_3 }
		}
		else_if = {
			limit = { from = { is_ai = no } OR = { is_country_type = dormant_marauders is_country_type = marauder_raiders is_country_type = awakened_marauders } }
			if = { limit = { has_country_flag = marauder_1 } from = { set_country_flag = sllm_fought_marauder_1 } }
			if = { limit = { has_country_flag = marauder_2 } from = { set_country_flag = sllm_fought_marauder_2 } }
			if = { limit = { has_country_flag = marauder_3 } from = { set_country_flag = sllm_fought_marauder_3 } }
		}
	}
}

# 매달: 칸의 총애를 받은 나라가 그레이트 칸의 태수국이면 칸의 총애 수정자를 붙이고, 아니면(독립·그레이트 칸 사망) 뗀다.
country_event = {
	id = sllm_khan.10
	hide_window = yes
	is_triggered_only = yes
	trigger = { is_ai = no OR = { has_country_flag = sllm_khan_favored has_modifier = sllm_khan_favor } }
	immediate = {
		if = {
			limit = { has_country_flag = sllm_khan_favored exists = overlord overlord = { is_country_type = awakened_marauders } }
			if = { limit = { NOT = { has_modifier = sllm_khan_favor } } add_modifier = { modifier = sllm_khan_favor days = -1 } log = "SLLM:KHANFAVOR/;/sid=\\[This.sllm_save_id]/;/on=1" }
		}
		else_if = {
			limit = { has_modifier = sllm_khan_favor }
			remove_modifier = sllm_khan_favor
			log = "SLLM:KHANFAVOR/;/sid=\\[This.sllm_save_id]/;/on=0"
		}
	}
}

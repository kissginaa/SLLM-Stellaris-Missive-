# 대화로 생긴 감사·원한은 이미 대화 결과(호감 변화)로 게임에 들어가므로 여기서 또 넣지 않는다(두 번 계산 방지).
sllm_emo_wary = {
	opinion = { base = -10 }
	unique = yes
}
sllm_emo_arrogant = {
	opinion = { base = -5 }
	unique = yes
}

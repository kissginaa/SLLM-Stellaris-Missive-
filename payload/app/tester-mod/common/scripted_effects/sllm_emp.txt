# 비상대권(2026-11-05) — 나라 scope. 변수 sllm_emp_lv(0~3) · sllm_emp_peace(전쟁 없이 지난 달 수) · sllm_emp_held(쥐고 지난 달 수)
# 표식 sllm_emp_cd(올린 뒤 180일은 다시 못 올림) · sllm_emp_hold(전쟁 뒤에도 쥐고 있음)
sllm_emp_apply = {
	remove_modifier = sllm_emp_1
	remove_modifier = sllm_emp_2
	remove_modifier = sllm_emp_3
	if = { limit = { check_variable = { which = sllm_emp_lv value = 1 } } add_modifier = { modifier = sllm_emp_1 days = -1 } }
	else_if = { limit = { check_variable = { which = sllm_emp_lv value = 2 } } add_modifier = { modifier = sllm_emp_2 days = -1 } }
	else_if = { limit = { check_variable = { which = sllm_emp_lv value >= 3 } } add_modifier = { modifier = sllm_emp_3 days = -1 } }
}
# 다 내려놓음: 전쟁이 끝난 뒤(석 달 이상 평화) 스스로 내려놓았으면 2년 안도
sllm_emp_end = {
	if = { limit = { check_variable = { which = sllm_emp_peace value >= 3 } } add_modifier = { modifier = sllm_emp_return days = 720 } }
	remove_modifier = sllm_emp_1
	remove_modifier = sllm_emp_2
	remove_modifier = sllm_emp_3
	remove_modifier = sllm_emp_grip
	remove_country_flag = sllm_emp_hold
	set_variable = { which = sllm_emp_lv value = 0 }
	clear_variable = sllm_emp_peace
	clear_variable = sllm_emp_held
}
sllm_emp_up = {
	change_variable = { which = sllm_emp_lv value = 1 }
	sllm_emp_apply = yes
	set_timed_country_flag = { flag = sllm_emp_cd days = 180 }
	set_variable = { which = sllm_emp_peace value = 0 }
}
sllm_emp_down = {
	subtract_variable = { which = sllm_emp_lv value = 1 }
	if = { limit = { check_variable = { which = sllm_emp_lv value <= 0 } } sllm_emp_end = yes }
	else = { sllm_emp_apply = yes }
}
sllm_emp_hold_on = {
	set_country_flag = sllm_emp_hold
	add_modifier = { modifier = sllm_emp_grip days = -1 }
	set_variable = { which = sllm_emp_held value = 0 }
}

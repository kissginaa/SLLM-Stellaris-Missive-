# 우리 기능의 남은 날 막대(②): TYPE = assim(동화 정책 다음 단계) | vatech(기술 원조 재사용), 대상 = event_target:sllm_al_t(종속국), IDX = 그 번호.
# 같은 종속국·같은 종류 막대는 하나만(먼저 치우고 새로). 막대가 다 차면 상황이 스스로 사라진다(sllm_alarm_situations.txt).
sllm_alarm_bar_off = {
	set_variable = { which = sllm_al_req value = $REQ$ }
	every_situation = {
		limit = { is_situation_type = sllm_alarm_$TYPE$ has_situation_flag = sllm_al_$IDX$ }
		destroy_situation = this
	}
	log = "SLLM:ALARMSET/;/req=\\[This.sllm_al_req]/;/sid=\\[This.sllm_save_id]/;/st=9"
}
sllm_alarm_bar_on = {
	set_variable = { which = sllm_al_req value = $REQ$ }
	every_situation = {
		limit = { is_situation_type = sllm_alarm_$TYPE$ has_situation_flag = sllm_al_$IDX$ }
		destroy_situation = this
	}
	if = {
		limit = { exists = event_target:sllm_al_t }
		start_situation = {
			type = sllm_alarm_$TYPE$
			target = event_target:sllm_al_t
			effect = {
				set_situation_flag = sllm_al_$IDX$
				set_variable = { which = sllm_al_total value = $DAYS$ }
			}
		}
	}
	log = "SLLM:ALARMSET/;/req=\\[This.sllm_al_req]/;/sid=\\[This.sllm_save_id]/;/st=9"
}

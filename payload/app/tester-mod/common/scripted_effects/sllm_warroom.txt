# ── 보고: 번호 없는 성계에 번호(새로 붙인 것만 WRSYS), 군 함대 목록(WRFL), 전쟁 상대(WRWAR), 합동 작전 상대(WRALLY) → WREND ──
sllm_wr_report = {
	set_variable = { which = sllm_wr_req value = $REQ$ }
	if = { limit = { NOT = { is_variable_set = sllm_sys_n } } set_variable = { which = sllm_sys_n value = 0 } }
	if = { limit = { NOT = { is_variable_set = sllm_fo_fidn } } set_variable = { which = sllm_fo_fidn value = 0 } }
	every_system = {
		limit = { exists = star star = { NOT = { is_variable_set = sllm_sys } } }
		star = {
			root = { change_variable = { which = sllm_sys_n value = 1 } }
			set_variable = { which = sllm_sys value = root.sllm_sys_n }
		}
		log = "SLLM:WRSYS/;/req=\\[Root.sllm_wr_req]/;/n=\\[Root.sllm_sys_n]/;/name=\\[This.GetName]"
	}
	if = { limit = { is_at_war = yes } set_variable = { which = sllm_wr_warv value = 1 } } else = { set_variable = { which = sllm_wr_warv value = 0 } }
	log = "SLLM:WRREP/;/req=\\[Root.sllm_wr_req]/;/war=\\[Root.sllm_wr_warv]/;/date=\\[GetDate]"
	every_controlled_fleet = {
		limit = { sllm_fo_military_fleet = yes }
		set_variable = { which = sllm_wr_aidx value = 0 }
		set_variable = { which = sllm_wr_xfv value = 0 }
		if = { limit = { exists = owner NOT = { owner = { is_same_value = root } } owner = { is_variable_set = sllm_idx } } owner = { prev = { set_variable = { which = sllm_wr_aidx value = prev.sllm_idx } } } }
		sllm_wr_fleet_line = yes
	}
	# 연합 원정군(sllm_warroom_xf.txt): 동맹이 보낸 함대를 가진, AI 가 꺼진 우리 쪽 세력 — 명령대로만 움직인다
	every_country = {
		limit = { has_country_flag = sllm_wr_xf }
		every_owned_fleet = {
			limit = { sllm_fo_military_fleet = yes }
			set_variable = { which = sllm_wr_aidx value = prev.sllm_wr_xf_ally }
			set_variable = { which = sllm_wr_xfv value = 1 }
			sllm_wr_fleet_line = yes
		}
	}
	every_country = {
		limit = { is_at_war_with = root }
		if = { limit = { is_variable_set = sllm_idx } set_variable = { which = sllm_wr_ci value = sllm_idx } } else = { set_variable = { which = sllm_wr_ci value = -1 } }
		log = "SLLM:WRWAR/;/req=\\[Root.sllm_wr_req]/;/idx=\\[This.sllm_wr_ci]/;/name=\\[This.GetName]"
	}
	# 합동 작전 상대: 관계(3 연방 · 2 방위 조약 · 1 같은 전쟁), 빌려 온 함대 수(n), 연합 원정군 함대 수(xf), 보낼 수 있는 함대 수·전력(ln·pow), 호감·신뢰 단계(op·tr), 같은 편 전쟁(cw), 함대 사령부 쪽 임대 중(ls)
	every_country = {
		limit = { sllm_fo_ally_ok = yes }
		if = { limit = { is_in_federation_with = root } set_variable = { which = sllm_wr_arel value = 3 } }
		else_if = { limit = { has_defensive_pact = root } set_variable = { which = sllm_wr_arel value = 2 } }
		else = { set_variable = { which = sllm_wr_arel value = 1 } }
		set_variable = { which = sllm_wr_an value = 0 }
		every_owned_fleet = { limit = { exists = controller controller = { is_same_value = root } } prev = { change_variable = { which = sllm_wr_an value = 1 } } }
		set_variable = { which = sllm_wr_axf value = 0 }
		save_event_target_as = sllm_wr_ac
		every_country = { limit = { has_country_flag = sllm_wr_xf check_variable = { which = sllm_wr_xf_ally value = prev.sllm_idx } } every_owned_fleet = { event_target:sllm_wr_ac = { change_variable = { which = sllm_wr_axf value = 1 } } } }
		if = { limit = { opinion = { who = root value >= 150 } } set_variable = { which = sllm_wr_aop value = 4 } }
		else_if = { limit = { opinion = { who = root value >= 75 } } set_variable = { which = sllm_wr_aop value = 3 } }
		else_if = { limit = { opinion = { who = root value >= 25 } } set_variable = { which = sllm_wr_aop value = 2 } }
		else_if = { limit = { opinion = { who = root value >= 0 } } set_variable = { which = sllm_wr_aop value = 1 } }
		else = { set_variable = { which = sllm_wr_aop value = 0 } }
		if = { limit = { trust = { who = root value >= 75 } } set_variable = { which = sllm_wr_atr value = 3 } }
		else_if = { limit = { trust = { who = root value >= 40 } } set_variable = { which = sllm_wr_atr value = 2 } }
		else_if = { limit = { trust = { who = root value >= 10 } } set_variable = { which = sllm_wr_atr value = 1 } }
		else = { set_variable = { which = sllm_wr_atr value = 0 } }
		set_variable = { which = sllm_wr_acw value = 0 }
		if = {
			limit = {
				any_war = {
					OR = {
						AND = { is_war_participant = { who = root side = attackers } is_war_participant = { who = prev side = attackers } }
						AND = { is_war_participant = { who = root side = defenders } is_war_participant = { who = prev side = defenders } }
					}
				}
			}
			set_variable = { which = sllm_wr_acw value = 1 }
		}
		set_variable = { which = sllm_wr_aln value = 0 }
		set_variable = { which = sllm_wr_apow value = 0 }
		every_owned_fleet = {
			limit = { sllm_fo_lendable_fleet = yes }
			set_variable = { which = sllm_wr_fp value = trigger:fleet_power }
			save_event_target_as = sllm_wr_af
			prev = { change_variable = { which = sllm_wr_aln value = 1 } change_variable = { which = sllm_wr_apow value = event_target:sllm_wr_af.sllm_wr_fp } }
		}
		if = { limit = { has_country_flag = sllm_fo_lending } set_variable = { which = sllm_wr_als value = 1 } } else = { set_variable = { which = sllm_wr_als value = 0 } }
		log = "SLLM:WRALLY/;/req=\\[Root.sllm_wr_req]/;/idx=\\[This.sllm_idx]/;/rel=\\[This.sllm_wr_arel]/;/n=\\[This.sllm_wr_an]/;/xf=\\[This.sllm_wr_axf]/;/ln=\\[This.sllm_wr_aln]/;/pow=\\[This.sllm_wr_apow]/;/op=\\[This.sllm_wr_aop]/;/tr=\\[This.sllm_wr_atr]/;/cw=\\[This.sllm_wr_acw]/;/ls=\\[This.sllm_wr_als]/;/name=\\[This.GetName]"
	}
	clear_variable = sllm_wr_cur
	log = "SLLM:WREND/;/req=\\[Root.sllm_wr_req]"
}

# ── 명령: FID 함대에 ACT(1 = SYS 번호 성계로 이동, 2 = 수도로 귀환, 3 = 멈춤) → WRSET ──
# 우리 함대: 직접 지휘 표식(자동 작전이 손대지 않음). 빌려 온 동맹 함대·연합 원정군: 「요청」 — 다쳤으면 거절, 가끔(12%) 자기 판단으로 거절, 받아들이면 120일만 우리 지휘.
sllm_wr_order_check = {
	if = {
		limit = { sllm_wr_foreign_fleet = yes NOT = { root = { check_variable = { which = sllm_wr_act value = 3 } } } }
		if = { limit = { has_hp_percentage < 0.5 } root = { set_variable = { which = sllm_wr_res value = 2 } } }
		else = { random_list = { 12 = { root = { set_variable = { which = sllm_wr_res value = 3 } } } 88 = { } } }
	}
}
sllm_wr_order_apply = {
	if = { limit = { sllm_wr_foreign_fleet = yes } set_timed_fleet_flag = { flag = sllm_fo_manual days = 120 } remove_fleet_flag = sllm_wr_busy }
	else = { set_fleet_flag = sllm_fo_manual }
	remove_fleet_flag = sllm_fo_lead
	remove_fleet_flag = sllm_fo_repair
	set_variable = { which = sllm_fo_op value = 0 }
	if = {
		limit = { root = { check_variable = { which = sllm_wr_act value = 3 } } }
		remove_auto_move_target = yes
		clear_fleet_actions = this
	}
	else = {
		auto_move_to_planet = { target = event_target:sllm_wr_dest clear_auto_move_on_arrival = yes }
	}
}
sllm_wr_order = {
	set_variable = { which = sllm_wr_req value = $REQ$ }
	set_variable = { which = sllm_wr_act value = $ACT$ }
	set_variable = { which = sllm_wr_sq value = $SYS$ }
	set_variable = { which = sllm_wr_res value = 0 }
	set_variable = { which = sllm_wr_found value = 0 }
	clear_global_event_target = sllm_wr_dest
	if = {
		limit = { check_variable = { which = sllm_wr_act value = 1 } }
		random_system = {
			limit = { exists = star star = { is_variable_set = sllm_sys check_variable = { which = sllm_sys value = root.sllm_wr_sq } } }
			star = { save_global_event_target_as = sllm_wr_dest }
		}
	}
	else_if = {
		limit = { check_variable = { which = sllm_wr_act value = 2 } exists = capital_scope }
		capital_star = { save_global_event_target_as = sllm_wr_dest }
	}
	every_controlled_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } root = { set_variable = { which = sllm_wr_found value = 1 } } }
	every_country = { limit = { has_country_flag = sllm_wr_xf } every_owned_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } root = { set_variable = { which = sllm_wr_found value = 1 } } } }
	if = {
		limit = { check_variable = { which = sllm_wr_found value = 0 } }
		log = "SLLM:WRSET/;/req=\\[Root.sllm_wr_req]/;/status=no_fleet"
	}
	else_if = {
		limit = { NOT = { check_variable = { which = sllm_wr_act value = 3 } } NOT = { exists = event_target:sllm_wr_dest } }
		log = "SLLM:WRSET/;/req=\\[Root.sllm_wr_req]/;/status=no_target"
	}
	else = {
		set_variable = { which = sllm_wr_res value = 1 }
		every_controlled_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } sllm_wr_order_check = yes }
		every_country = { limit = { has_country_flag = sllm_wr_xf } every_owned_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } sllm_wr_order_check = yes } }
	}
	if = {
		limit = { check_variable = { which = sllm_wr_res value = 2 } }
		log = "SLLM:WRSET/;/req=\\[Root.sllm_wr_req]/;/status=ally_repair"
	}
	else_if = {
		limit = { check_variable = { which = sllm_wr_res value = 3 } }
		log = "SLLM:WRSET/;/req=\\[Root.sllm_wr_req]/;/status=ally_refuse"
	}
	else_if = {
		limit = { check_variable = { which = sllm_wr_res value = 1 } }
		every_controlled_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } sllm_wr_order_apply = yes }
		every_country = { limit = { has_country_flag = sllm_wr_xf } every_owned_fleet = { limit = { sllm_wr_fid_is = { FID = $FID$ } } sllm_wr_order_apply = yes } }
		log = "SLLM:WRSET/;/req=\\[Root.sllm_wr_req]/;/status=applied"
	}
	clear_global_event_target = sllm_wr_dest
	set_variable = { which = sllm_wr_res value = 0 }
}

# ── 빌려 온 동맹 함대의 독자 행동(매달, sllm_wr.1). 우리 자동 작전에 들어가 있거나(sllm_fo_op > 0) 우리 지휘를 받는 중(sllm_fo_manual)이면 손대지 않는다. ──
# 쉬고 있는 동맹 함대마다: 다쳤으면 가장 가까운 우리 조선소로(EV 3), 전쟁 중이면 10칸 안의 가장 가까운 적 성계로(적 식민지 먼저, EV 1), 평시엔 가끔 우리 국경 순찰(EV 2).
# 연합 원정군 함대(sllm_wr_xff)도 같은 규칙으로 — 그쪽은 AI 가 꺼져 있어 정한 그대로 간다.
# 한 번 움직이면 60일 동안 다시 정하지 않는다(sllm_wr_busy). 로그 WRAUTO(함대) + WRAUTOS(목적지), 같은 n.
# 함대 한 개의 판단(함대 범위, sllm_wr_aidx 를 미리 정해 둠)
sllm_wr_auto_fleet = {
	clear_global_event_target = sllm_wr_t
	set_variable = { which = sllm_wr_ev value = 0 }
	if = {
		limit = { has_hp_percentage < 0.5 }
		solar_system = { closest_system = { min_steps = 0 max_steps = 40 limit = { any_starbase_in_system = { exists = owner owner = { is_same_value = root } has_starbase_module = shipyard } } star = { save_global_event_target_as = sllm_wr_t } } }
		set_variable = { which = sllm_wr_ev value = 3 }
	}
	else_if = {
		limit = { root = { is_at_war = yes } }
		solar_system = { closest_system = { min_steps = 0 max_steps = 10 limit = { sllm_fo_war_target = yes sllm_fo_war_colony = yes } star = { save_global_event_target_as = sllm_wr_t } } }
		if = { limit = { NOT = { exists = event_target:sllm_wr_t } } solar_system = { closest_system = { min_steps = 0 max_steps = 10 limit = { sllm_fo_war_target = yes } star = { save_global_event_target_as = sllm_wr_t } } } }
		set_variable = { which = sllm_wr_ev value = 1 }
	}
	else = {
		random = { chance = 25
			root = { random_system_within_border = { limit = { exists = star any_neighbor_system = { exists = owner NOT = { owner = { is_same_value = root } } } } star = { save_global_event_target_as = sllm_wr_t } } }
			set_variable = { which = sllm_wr_ev value = 2 }
		}
	}
	if = {
		limit = { exists = event_target:sllm_wr_t check_variable = { which = sllm_wr_ev value > 0 } }
		auto_move_to_planet = { target = event_target:sllm_wr_t clear_auto_move_on_arrival = yes }
		if = { limit = { check_variable = { which = sllm_wr_ev value = 3 } } set_fleet_stance = passive } else = { set_fleet_stance = aggressive }
		set_timed_fleet_flag = { flag = sllm_wr_busy days = 60 }
		root = { change_variable = { which = sllm_wr_evn value = 1 } }
		log = "SLLM:WRAUTO/;/n=\\[Root.sllm_wr_evn]/;/ev=\\[This.sllm_wr_ev]/;/ally=\\[This.sllm_wr_aidx]/;/date=\\[GetDate]/;/fleet=\\[This.GetName]/;/adm=\\[This.GetLeaderName]/;/at=\\[This.System.GetName]"
		event_target:sllm_wr_t = { solar_system = { log = "SLLM:WRAUTOS/;/n=\\[Root.sllm_wr_evn]/;/sys=\\[This.GetName]" } }
	}
	clear_global_event_target = sllm_wr_t
}
sllm_wr_ally_tick = {
	if = { limit = { NOT = { is_variable_set = sllm_wr_evn } } set_variable = { which = sllm_wr_evn value = 0 } }
	every_controlled_fleet = {
		limit = { sllm_wr_ally_fleet = yes sllm_wr_auto_ok = yes }
		set_variable = { which = sllm_wr_aidx value = 0 }
		if = { limit = { owner = { is_variable_set = sllm_idx } } owner = { prev = { set_variable = { which = sllm_wr_aidx value = prev.sllm_idx } } } }
		sllm_wr_auto_fleet = yes
	}
	every_country = {
		limit = { has_country_flag = sllm_wr_xf }
		every_owned_fleet = {
			limit = { sllm_fo_military_fleet = yes sllm_wr_auto_ok = yes }
			set_variable = { which = sllm_wr_aidx value = prev.sllm_wr_xf_ally }
			sllm_wr_auto_fleet = yes
		}
	}
}

# SLLM 공동 대응 협약(알파 11, 2026-10-25, 위기 제안 5번). 앱: src/crisis-pact.ts · src/features/crisis-pact-app.ts.
# 수치(사용자 확정): 서로의 호감 +30(sllm_opinion_crisis_pact), 양쪽 함선 화력 +5%(sllm_crisis_pact). 협약 상대 = 국가 플래그 sllm_pact_partner.
# 맺기(앱이 대화 제안을 플레이어가 수락했을 때): effect every_playable_country = { limit = { is_ai = no } sllm_crisis_pact_make = { REQ = 9 TID = 2 } }
sllm_crisis_pact_make = {
	save_event_target_as = sllm_player
	sllm_probe_init_vars = yes
	set_variable = { which = sllm_in_req value = $REQ$ }
	if = {
		limit = { check_variable = { which = sllm_last_req value < $REQ$ } }
		set_variable = { which = sllm_last_req value = $REQ$ }
		change_variable = { which = sllm_exec_seq value = 1 }
		set_variable = { which = sllm_last_result value = 2 }
		random_country = {
			limit = {
				is_variable_set = sllm_idx
				check_variable = { which = sllm_idx value = $TID$ }
				NOT = { is_at_war_with = event_target:sllm_player }
			}
			set_country_flag = sllm_pact_partner
			remove_modifier = sllm_crisis_pact
			add_modifier = { modifier = sllm_crisis_pact days = -1 }
			remove_opinion_modifier = { who = event_target:sllm_player modifier = sllm_opinion_crisis_pact }
			add_opinion_modifier = { who = event_target:sllm_player modifier = sllm_opinion_crisis_pact }
			event_target:sllm_player = {
				remove_opinion_modifier = { who = prev modifier = sllm_opinion_crisis_pact }
				add_opinion_modifier = { who = prev modifier = sllm_opinion_crisis_pact }
				set_variable = { which = sllm_last_result value = 1 }
			}
		}
		if = {
			limit = { check_variable = { which = sllm_last_result value = 1 } }
			remove_modifier = sllm_crisis_pact
			add_modifier = { modifier = sllm_crisis_pact days = -1 }
		}
		log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/pact/;/result=\\[This.sllm_last_result]"
	}
	else_if = {
		limit = { check_variable = { which = sllm_last_req value = $REQ$ } }
		log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/pact/;/dup/;/result=\\[This.sllm_last_result]"
	}
	else = {
		log = "SLLM:ACK/;/seq=\\[This.sllm_exec_seq]/;/req=\\[This.sllm_in_req]/;/pact/;/stale/;/last=\\[This.sllm_last_req]"
	}
}
# 끝내기(위기가 끝났거나 협약 중 전쟁): 그 상대의 화력·호감을 지우고, 남은 상대가 없으면 플레이어의 화력도 지운다. 여러 번 보내도 안전.
# effect every_playable_country = { limit = { is_ai = no } save_event_target_as = sllm_player sllm_crisis_pact_end = { TID = 2 } }
sllm_crisis_pact_end = {
	random_country = {
		limit = {
			is_variable_set = sllm_idx
			check_variable = { which = sllm_idx value = $TID$ }
		}
		remove_country_flag = sllm_pact_partner
		remove_modifier = sllm_crisis_pact
		remove_opinion_modifier = { who = event_target:sllm_player modifier = sllm_opinion_crisis_pact }
		event_target:sllm_player = { remove_opinion_modifier = { who = prev modifier = sllm_opinion_crisis_pact } }
	}
	if = {
		limit = { NOT = { any_country = { has_country_flag = sllm_pact_partner } } }
		remove_modifier = sllm_crisis_pact
	}
}

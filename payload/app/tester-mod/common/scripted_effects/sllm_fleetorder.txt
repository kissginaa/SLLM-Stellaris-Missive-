# SLLM 자동 작전(2026-11-04) — 함대 사령부 「자동 작전」. 앱이 켜고 끈다(국가 표식 sllm_fo_on). 우리가 직접 쓴 스크립트(설계만 「자동작전 · Auto Combat」 모드 참고).
# 매달(on_monthly_pulse_country → sllm_fo.1): 수리 복귀 → 작전 1~3 진행. 전쟁 중일 때만 새 작전을 연다.
# 작전 하나 = 선두 함대(가장 센 함대) + 불러온 함대. 목표 = 수도에서 가장 가까운, 우리 전력(÷1.2)으로 이길 만한 적 성계(적 식민지 성계 먼저).
#   상태 sllm_fo_opN: 0 비어 있음, 1 집결(선두 성계로 모임, 최대 5달), 2 진격. 적이 우리보다 세지면 후퇴, 적이 없어지면 완료.
# 작전 함대 풀은 every_controlled_fleet: 우리 함대 + 빌려 온 동맹 함대(give_fleet, sllm_fleetorder_ally.txt). 우리가 빌려준 함대는 빠진다.
# 함대 변수·표식: sllm_fo_op(0/1~3), sllm_fo_lead(선두), sllm_fo_manual(직접 지휘 — 손대지 않음), sllm_fo_repair(수리하러 감), sllm_fo_fid(앱이 쓰는 번호).
# 로그(앱이 읽음): FOEV(사건) + FOEVF(그 함대·제독) + FOEVS(성계) — 같은 n 끼리 한 묶음. FOREP·FOOP·FOOPF·FOOPS·FOFL(보고), FOSET(켜기·끄기·전환 응답).
# 게임 부담: 작전마다 한 달에 함대 목록 몇 번 + 빈 작전일 때만 수도 둘레 40칸 안에서 가장 가까운 성계 찾기.

# ── 작은 도우미 ──
# 사건 번호를 올린다(로그 묶음 n).
sllm_fo_ev_next = {
	if = { limit = { NOT = { is_variable_set = sllm_fo_evn } } set_variable = { which = sllm_fo_evn value = 0 } }
	change_variable = { which = sllm_fo_evn value = 1 }
}
# 작전 N 의 함대 전력 합(sllm_fo_pow)과 함대 수(sllm_fo_n).
sllm_fo_op_power = {
	set_variable = { which = sllm_fo_pow value = 0 }
	set_variable = { which = sllm_fo_n value = 0 }
	every_controlled_fleet = {
		limit = { is_variable_set = sllm_fo_op check_variable = { which = sllm_fo_op value = $N$ } }
		set_variable = { which = sllm_fo_fp value = trigger:fleet_power }
		save_event_target_as = sllm_fo_f
		root = {
			change_variable = { which = sllm_fo_pow value = event_target:sllm_fo_f.sllm_fo_fp }
			change_variable = { which = sllm_fo_n value = 1 }
		}
	}
}
# 작전 N 의 목표 성계 적 전력(sllm_fo_en). 엔진 값이 안 읽히면 고를 때의 단계 값으로.
sllm_fo_op_enemy = {
	set_variable = { which = sllm_fo_en value = sllm_fo_op$N$_tier }
	event_target:sllm_fo_op$N$_sys = {
		export_trigger_value_to_variable = { trigger = hostile_military_power parameters = { who = root safety_buffer = no } variable = sllm_fo_en }
		if = {
			limit = { is_variable_set = sllm_fo_en check_variable = { which = sllm_fo_en value > 0 } }
			root = { set_variable = { which = sllm_fo_en value = event_target:sllm_fo_op$N$_sys.sllm_fo_en } }
		}
	}
}
# 작전 N 의 함대를 모두 목표 성계로 보낸다(이미 그 성계에 있거나 전투 중이면 그대로).
sllm_fo_op_send = {
	every_controlled_fleet = {
		limit = {
			is_variable_set = sllm_fo_op
			check_variable = { which = sllm_fo_op value = $N$ }
			is_in_combat = no
			NOT = { solar_system = { is_same_value = event_target:sllm_fo_op$N$_sys } }
		}
		auto_move_to_planet = { target = event_target:sllm_fo_op$N$_star clear_auto_move_on_arrival = yes }
	}
}
# 사건 로그: 선두 함대·제독, 목표 성계. EVN: 1 new 2 go 3 done 4 lost 5 lead 6 abort 7 retreat (8 repair·9 repaired 는 수리 쪽). 로그 글자 안에는 $매개변수$ 를 쓰지 않는다(변수로 옮겨 찍는다 — 게임이 글자 속 매개변수를 잘못 바꾼 적이 있음).
sllm_fo_op_log = {
	sllm_fo_ev_next = yes
	set_variable = { which = sllm_fo_lop value = $N$ }
	set_variable = { which = sllm_fo_lev value = $EVN$ }
	log = "SLLM:FOEV/;/n=\\[Root.sllm_fo_evn]/;/ev=\\[Root.sllm_fo_lev]/;/op=\\[Root.sllm_fo_lop]/;/date=\\[GetDate]/;/pow=\\[Root.sllm_fo_pow]/;/en=\\[Root.sllm_fo_en]/;/cnt=\\[Root.sllm_fo_n]"
	if = {
		limit = { exists = event_target:sllm_fo_op$N$_lead }
		event_target:sllm_fo_op$N$_lead = {
			log = "SLLM:FOEVF/;/n=\\[Root.sllm_fo_evn]/;/fleet=\\[This.GetName]/;/adm=\\[This.GetLeaderName]/;/at=\\[This.System.GetName]"
			if = { limit = { exists = leader leader = { is_variable_set = sllm_lid } } leader = { log = "SLLM:FOEVA/;/n=\\[Root.sllm_fo_evn]/;/lid=\\[This.sllm_lid]" } }
		}
	}
	if = { limit = { exists = event_target:sllm_fo_op$N$_sys } event_target:sllm_fo_op$N$_sys = { log = "SLLM:FOEVS/;/n=\\[Root.sllm_fo_evn]/;/sys=\\[This.GetName]" } }
}
# 작전 N 끝: 함대를 풀어 주고 목표 표식을 지운다. HOME = yes 면 수도로 물린다(후퇴·중단).
sllm_fo_op_end = {
	every_controlled_fleet = {
		limit = { is_variable_set = sllm_fo_op check_variable = { which = sllm_fo_op value = $N$ } }
		set_variable = { which = sllm_fo_op value = 0 }
		remove_fleet_flag = sllm_fo_lead
		if = {
			limit = { always = $HOME$ is_in_combat = no }
			root = { capital_star = { save_event_target_as = sllm_fo_home_star } }
			if = { limit = { exists = event_target:sllm_fo_home_star } auto_move_to_planet = { target = event_target:sllm_fo_home_star clear_auto_move_on_arrival = yes } }
		}
	}
	if = { limit = { exists = event_target:sllm_fo_op$N$_sys } event_target:sllm_fo_op$N$_sys = { remove_star_flag = sllm_fo_tgt } }
	clear_global_event_target = sllm_fo_op$N$_sys
	clear_global_event_target = sllm_fo_op$N$_star
	clear_global_event_target = sllm_fo_op$N$_lead
	set_variable = { which = sllm_fo_op$N$ value = 0 }
	set_variable = { which = sllm_fo_op$N$_m value = 0 }
}
# 자유 함대 가운데 가장 센 것을 작전 N 에 넣고 선두 성계로 부른다.
sllm_fo_op_recruit = {
	if = {
		limit = {
			check_variable_arithmetic = { which = sllm_fo_en multiply = sllm_fo_ratio value > sllm_fo_pow }
			any_controlled_fleet = { sllm_fo_free_fleet = yes }
		}
		ordered_controlled_fleet = {
			limit = { sllm_fo_free_fleet = yes }
			order_by = trigger:fleet_power
			position = 0
			set_variable = { which = sllm_fo_op value = $N$ }
			set_fleet_stance = aggressive
			set_variable = { which = sllm_fo_fp value = trigger:fleet_power }
			save_event_target_as = sllm_fo_f
			root = {
				change_variable = { which = sllm_fo_pow value = event_target:sllm_fo_f.sllm_fo_fp }
				change_variable = { which = sllm_fo_n value = 1 }
			}
			if = {
				limit = { exists = event_target:sllm_fo_op$N$_lead }
				event_target:sllm_fo_op$N$_lead = { solar_system = { star = { save_event_target_as = sllm_fo_rally } } }
				if = { limit = { exists = event_target:sllm_fo_rally } auto_move_to_planet = { target = event_target:sllm_fo_rally clear_auto_move_on_arrival = yes } }
			}
		}
	}
}

# ── 목표 찾기: 우리 전력 ÷ 1.2 아래의 단계(MAX)에서, 수도에서 가장 가까운 적 성계 ──
sllm_fo_find = {
	if = {
		limit = { check_variable = { which = sllm_fo_found value = 0 } check_variable = { which = sllm_fo_cap value >= $MAX$ } }
		# 1차: 적 식민지가 있는 성계
		capital_scope = {
			solar_system = {
				closest_system = {
					min_steps = 0
					max_steps = $STEPS$
					limit = { sllm_fo_war_target = yes sllm_fo_war_colony = yes hostile_military_power = { who = root safety_buffer = no value <= $MAX$ } }
					save_global_event_target_as = sllm_fo_op$N$_sys
					star = { save_global_event_target_as = sllm_fo_op$N$_star }
					root = { set_variable = { which = sllm_fo_found value = 1 } }
				}
			}
		}
		# 2차: 그 밖의 적 성계(적 정거장·함대)
		if = {
			limit = { check_variable = { which = sllm_fo_found value = 0 } }
			capital_scope = {
				solar_system = {
					closest_system = {
						min_steps = 0
						max_steps = $STEPS$
						limit = { sllm_fo_war_target = yes hostile_military_power = { who = root safety_buffer = no value <= $MAX$ } }
						save_global_event_target_as = sllm_fo_op$N$_sys
						star = { save_global_event_target_as = sllm_fo_op$N$_star }
						root = { set_variable = { which = sllm_fo_found value = 1 } }
					}
				}
			}
		}
		if = { limit = { check_variable = { which = sllm_fo_found value = 1 } } set_variable = { which = sllm_fo_op$N$_tier value = $MAX$ } }
		# 단계 하나를 봤으면 더 낮은 단계는 보지 않는다(같은 달에 성계 찾기를 여러 번 하지 않게)
		set_variable = { which = sllm_fo_found value = 2 }
	}
}

sllm_fo_op_new = {
	if = {
		limit = { is_at_war = yes exists = capital_scope check_variable = { which = sllm_fo_maxops value >= $N$ } any_controlled_fleet = { sllm_fo_free_fleet = yes } }
		# 쓸 수 있는 전력 = 자유 함대 전부
		set_variable = { which = sllm_fo_cap value = 0 }
		every_controlled_fleet = {
			limit = { sllm_fo_free_fleet = yes }
			set_variable = { which = sllm_fo_fp value = trigger:fleet_power }
			save_event_target_as = sllm_fo_f
			root = { change_variable = { which = sllm_fo_cap value = event_target:sllm_fo_f.sllm_fo_fp } }
		}
		divide_variable = { which = sllm_fo_cap value = sllm_fo_ratio }
		set_variable = { which = sllm_fo_found value = 0 }
		# 작전 지시의 우선 목표가 칠 수 있는 곳이면 그곳부터(전력이 모자라면 집결하며 더 불러 모은다)
		if = {
			limit = { exists = event_target:sllm_fo_prio_sys event_target:sllm_fo_prio_sys = { sllm_fo_war_target = yes } }
			event_target:sllm_fo_prio_sys = {
				save_global_event_target_as = sllm_fo_op$N$_sys
				star = { save_global_event_target_as = sllm_fo_op$N$_star }
			}
			set_variable = { which = sllm_fo_op$N$_tier value = sllm_fo_cap }
			set_variable = { which = sllm_fo_found value = 2 }
		}
		# 작전 반경(지시): 1 가까이 12칸 · 2 중간 25칸 · 3 멀리 40칸
		if = { limit = { check_variable = { which = sllm_fo_range value = 1 } } sllm_fo_find = { N = $N$ MAX = 400000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 200000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 100000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 60000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 30000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 15000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 8000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 4000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 2000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 1000 STEPS = 12 } sllm_fo_find = { N = $N$ MAX = 500 STEPS = 12 } }
		else_if = { limit = { check_variable = { which = sllm_fo_range value = 2 } } sllm_fo_find = { N = $N$ MAX = 400000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 200000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 100000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 60000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 30000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 15000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 8000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 4000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 2000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 1000 STEPS = 25 } sllm_fo_find = { N = $N$ MAX = 500 STEPS = 25 } }
		else = { sllm_fo_find = { N = $N$ MAX = 400000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 200000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 100000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 60000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 30000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 15000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 8000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 4000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 2000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 1000 STEPS = 40 } sllm_fo_find = { N = $N$ MAX = 500 STEPS = 40 } }
		if = {
			limit = { exists = event_target:sllm_fo_op$N$_sys }
			event_target:sllm_fo_op$N$_sys = { set_star_flag = sllm_fo_tgt }
			# 선두 = 가장 센 자유 함대
			ordered_controlled_fleet = {
				limit = { sllm_fo_free_fleet = yes }
				order_by = trigger:fleet_power
				position = 0
				set_variable = { which = sllm_fo_op value = $N$ }
				set_fleet_flag = sllm_fo_lead
				set_fleet_stance = aggressive
				save_global_event_target_as = sllm_fo_op$N$_lead
			}
			set_variable = { which = sllm_fo_op$N$ value = 1 }
			set_variable = { which = sllm_fo_op$N$_m value = 0 }
			sllm_fo_op_power = { N = $N$ }
			sllm_fo_op_enemy = { N = $N$ }
			sllm_fo_op_log = { N = $N$ EVN = 1 }

}

# ── 끝까지 추격(2026-11-04, 「Search and Destroy」의 추격을 참고해 우리가 씀): 작전 지시 sllm_fo_chase = 1 이고 진격 중(2)일 때만.
# 지금 목표 성계에서 3칸 안, 적 군 함대가 보이는 칠 만한 성계(sllm_fo_war_target) 가운데 가장 가까운 곳으로 목표를 옮긴다. 찾으면 sllm_fo_chased = 1.
sllm_fo_op_chase = {
	set_variable = { which = sllm_fo_chased value = 0 }
	if = {
		limit = {
			check_variable = { which = sllm_fo_chase value = 1 }
			check_variable = { which = sllm_fo_op$N$ value = 2 }
			exists = event_target:sllm_fo_op$N$_sys
		}
		event_target:sllm_fo_op$N$_sys = {
			remove_star_flag = sllm_fo_tgt
			closest_system = {
				min_steps = 1
				max_steps = 3
				limit = {
					sllm_fo_war_target = yes
					any_fleet_in_system = { exists = owner owner = { is_at_war_with = root } is_civilian = no is_mobile = yes NOT = { is_cloaked = yes } }
				}
				save_global_event_target_as = sllm_fo_op$N$_sys
				star = { save_global_event_target_as = sllm_fo_op$N$_star }
				root = { set_variable = { which = sllm_fo_chased value = 1 } }
			}
		}
		if = { limit = { exists = event_target:sllm_fo_op$N$_sys } event_target:sllm_fo_op$N$_sys = { set_star_flag = sllm_fo_tgt } }
	}
}

sllm_fo_op_tick = {
	if = { limit = { NOT = { is_variable_set = sllm_fo_op$N$ } } set_variable = { which = sllm_fo_op$N$ value = 0 } }
	if = {
		limit = { check_variable = { which = sllm_fo_op$N$ value > 0 } }
		change_variable = { which = sllm_fo_op$N$_m value = 1 }
		sllm_fo_op_power = { N = $N$ }
		if = {
			# 목표가 사라졌거나 적이 없다 → 완료
			limit = {
				OR = {
					NOT = { exists = event_target:sllm_fo_op$N$_sys }
					event_target:sllm_fo_op$N$_sys = { NOT = { hostile_military_power = { who = root safety_buffer = no value > 0 } } }
				}
			}
			# 끝까지 추격(2026-11-04): 진격 중에 적 함대가 빠져나갔으면 3칸 안의 적 함대 성계로 목표를 옮겨 계속 쫓는다(세면 다음 달 기존 규칙대로 물러남)
			sllm_fo_op_chase = { N = $N$ }
			if = {
				limit = { check_variable = { which = sllm_fo_chased value = 1 } }
				sllm_fo_op_send = { N = $N$ }
				sllm_fo_op_log = { N = $N$ EVN = 2 }
			}
			else = {
				set_variable = { which = sllm_fo_en value = 0 }
				sllm_fo_op_log = { N = $N$ EVN = 3 }
				if = { limit = { exists = event_target:sllm_fo_prio_sys exists = event_target:sllm_fo_op$N$_sys event_target:sllm_fo_prio_sys = { is_same_value = event_target:sllm_fo_op$N$_sys } } clear_global_event_target = sllm_fo_prio_sys }
				sllm_fo_op_end = { N = $N$ HOME = no }
			}
		}
		else_if = {
			# 함대가 하나도 안 남았다(전멸·전부 수리)
			limit = { check_variable = { which = sllm_fo_n value = 0 } }
			set_variable = { which = sllm_fo_en value = 0 }
			sllm_fo_op_log = { N = $N$ EVN = 4 }
			sllm_fo_op_end = { N = $N$ HOME = no }
		}
		else = {
			sllm_fo_op_enemy = { N = $N$ }
			# 선두가 빠졌으면(파괴·수리·직접 지휘) 남은 함대 가운데 가장 센 함대가 지휘를 넘겨받는다
			if = {
				limit = {
					NOT = { any_controlled_fleet = { has_fleet_flag = sllm_fo_lead is_variable_set = sllm_fo_op check_variable = { which = sllm_fo_op value = $N$ } } }
				}
				ordered_controlled_fleet = {
					limit = { is_variable_set = sllm_fo_op check_variable = { which = sllm_fo_op value = $N$ } }
					order_by = trigger:fleet_power
					position = 0
					set_fleet_flag = sllm_fo_lead
					save_global_event_target_as = sllm_fo_op$N$_lead
				}
				sllm_fo_op_log = { N = $N$ EVN = 5 }
			}
			if = {
				# 집결
				limit = { check_variable = { which = sllm_fo_op$N$ value = 1 } }
				sllm_fo_op_recruit = { N = $N$ }
				sllm_fo_op_recruit = { N = $N$ }
				if = {
					limit = {
						check_variable_arithmetic = { which = sllm_fo_en multiply = sllm_fo_ratio value <= sllm_fo_pow }
						OR = {
							check_variable = { which = sllm_fo_op$N$_m value >= 5 }
							NOT = {
								any_controlled_fleet = {
									is_variable_set = sllm_fo_op
									check_variable = { which = sllm_fo_op value = $N$ }
									NOT = { solar_system = { is_same_value = event_target:sllm_fo_op$N$_lead.solar_system } }
								}
							}
						}
					}
					set_variable = { which = sllm_fo_op$N$ value = 2 }
					set_variable = { which = sllm_fo_op$N$_m value = 0 }
					sllm_fo_op_send = { N = $N$ }
					sllm_fo_op_log = { N = $N$ EVN = 2 }
				}
				else_if = {
					# 다섯 달을 기다려도 모자라다 → 접는다
					limit = { check_variable = { which = sllm_fo_op$N$_m value >= 5 } }
					sllm_fo_op_log = { N = $N$ EVN = 6 }
					sllm_fo_op_end = { N = $N$ HOME = no }
				}
			}
			else = {
				# 진격: 적이 우리보다 세지면 수도로 물린다. 1년 반이 지나도 안 끝나면 접는다. 아니면 늦게 온 함대도 보낸다.
				if = {
					limit = { check_variable_arithmetic = { which = sllm_fo_en value > sllm_fo_pow } }
					sllm_fo_op_log = { N = $N$ EVN = 7 }
					sllm_fo_op_end = { N = $N$ HOME = yes }
				}
				else_if = {
					limit = { check_variable = { which = sllm_fo_op$N$_m value >= 18 } }
					sllm_fo_op_log = { N = $N$ EVN = 6 }
					sllm_fo_op_end = { N = $N$ HOME = no }
				}
				else = {
					sllm_fo_op_send = { N = $N$ }
				}
			}
		}
	}
	else = {
		sllm_fo_op_new = { N = $N$ }
	}
}

sllm_mt_pay_1_1 = {
	if = {
		limit = { check_variable = { which = sllm_mt1_d value = 1 } }
		event_target:sllm_mt_p = { export_resource_stockpile_to_variable = { resource = energy variable = sllm_mt_stock } set_variable = { which = sllm_mt_amt value = prev.sllm_mt1_a } }
		if = {
			limit = { event_target:sllm_mt_p = { check_variable = { which = sllm_mt_stock value >= sllm_mt_amt } } }
			event_target:sllm_mt_p = { add_resource = { energy = -1 mult = sllm_mt_amt } }
			add_resource = { energy = 1 mult = sllm_mt1_a }
			set_variable = { which = sllm_mt_ok value = 1 }
		}
	}
	else = {
		export_resource_stockpile_to_variable = { resource = energy variable = sllm_mt_stock }
		if = {
			limit = { check_variable = { which = sllm_mt_stock value >= sllm_mt1_a } }
			add_resource = { energy = -1 mult = sllm_mt1_a }
			event_target:sllm_mt_p = { set_variable = { which = sllm_mt_amt value = prev.sllm_mt1_a } add_resource = { energy = 1 mult = sllm_mt_amt } }
			set_variable = { which = sllm_mt_ok value = 1 }
		}
	}
}

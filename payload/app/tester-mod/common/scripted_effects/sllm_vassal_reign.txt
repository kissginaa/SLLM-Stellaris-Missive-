sllm_vr_leaders = {

sllm_vr_leader_line = {

	set_variable = { which = sllm_vr_ln value = root.sllm_vr_lc }
	set_variable = { which = sllm_vr_lk value = 0 }
	set_variable = { which = sllm_vr_ls value = $SIDE$ } # 1 = 종속국 사람(t) · 2 = 우리 사람(me) — 로그 문자열엔 매개변수를 넣지 않고 변수로(F-39)

	log = "SLLM:VRLDR/;/req=\\[Root.sllm_vr_req]/;/n=\\[This.sllm_vr_ln]/;/side=\\[This.sllm_vr_ls]/;/cls=\\[This.sllm_vr_lk]/;/name=\\[This.GetName]"
}
# 앉히기: OWNER = sllm_vr_t(종속국 사람) / sllm_vr_me(우리 사람, MOVE = yes 면 그 나라로 넘긴다). MOD = 새 통치자 수정자.
sllm_vr_install = {
	set_variable = { which = sllm_vr_req value = $REQ$ }
	set_variable = { which = sllm_vr_st value = 0 }
	event_target:$OWNER$ = {
		random_owned_leader = {
			limit = { is_variable_set = sllm_vr_ln check_variable = { which = sllm_vr_ln value = $N$ } is_ruler = no }
			save_event_target_as = sllm_vr_pick
		}
	}
	if = {
		limit = { exists = event_target:sllm_vr_pick }
		if = {
			limit = { event_target:sllm_vr_t = { has_country_flag = sllm_vr_install_cd } }
			set_variable = { which = sllm_vr_st value = 7 }
		}
		else_if = {
			limit = { resource_stockpile_compare = { resource = influence value < $COST$ } }
			set_variable = { which = sllm_vr_st value = 3 }
		}
		else = {
			event_target:sllm_vr_t = { ruler = { save_event_target_as = sllm_vr_old } }
			if = { limit = { always = $MOVE$ } event_target:sllm_vr_pick = { set_owner = event_target:sllm_vr_t } }
			event_target:sllm_vr_t = { assign_leader = event_target:sllm_vr_pick }
			if = {
				limit = { event_target:sllm_vr_t = { ruler = { is_same_value = event_target:sllm_vr_pick } } }
				add_resource = { influence = -$COST$ }
				event_target:sllm_vr_t = {
					add_modifier = { modifier = $MOD$ days = $DAYS$ }
					add_modifier = { modifier = sllm_vr_grudge days = 1080 }
					set_timed_country_flag = { flag = sllm_vr_install_cd days = 3600 }
				}
				if = {
					limit = { exists = event_target:sllm_vr_old }
					event_target:sllm_vr_old = {
						set_leader_flag = sllm_vr_deposed
						log = "SLLM:VRNAME/;/req=\\[Root.sllm_vr_req]/;/k=old/;/name=\\[This.GetName]"
					}
				}
				event_target:sllm_vr_pick = {
					set_leader_flag = sllm_vr_puppet_ruler
					log = "SLLM:VRNAME/;/req=\\[Root.sllm_vr_req]/;/k=new/;/name=\\[This.GetName]"
				}
				set_variable = { which = sllm_vr_st value = 9 }
			}
			else = {
				if = { limit = { always = $MOVE$ } event_target:sllm_vr_pick = { set_owner = event_target:sllm_vr_me } }
				set_variable = { which = sllm_vr_st value = 6 }
			}
		}
	}
	log = "SLLM:VRSET/;/req=\\[This.sllm_vr_req]/;/sid=\\[This.sllm_save_id]/;/st=\\[This.sllm_vr_st]"
}

sllm_vr_targets = {

	every_country = {

# 선포: COST = 영향력(우리), E = 지원금(우리 → 종속국 에너지). 대상에는 sllm_vr_px_<종속국 번호> 표식(월 점검용).
sllm_vr_proxy = {
	set_variable = { which = sllm_vr_req value = $REQ$ }
	set_variable = { which = sllm_vr_st value = 0 }
	random_country = {
		limit = { is_variable_set = sllm_vr_gn check_variable = { which = sllm_vr_gn value = $N$ } NOT = { is_at_war_with = event_target:sllm_vr_t } }
		save_event_target_as = sllm_vr_g
	}
	if = {
		limit = { exists = event_target:sllm_vr_g }
		if = {
			limit = { resource_stockpile_compare = { resource = influence value < $COST$ } }
			set_variable = { which = sllm_vr_st value = 3 }
		}
		else_if = {
			limit = { resource_stockpile_compare = { resource = energy value < $E$ } }
			set_variable = { which = sllm_vr_st value = 4 }
		}
		else = {
			add_resource = { influence = -$COST$ energy = -$E$ }
			event_target:sllm_vr_g = { set_country_flag = sllm_vr_px_$TID$ set_country_flag = sllm_vr_pxt set_variable = { which = sllm_vr_pxv value = $TID$ } } # pxt·pxv: 전쟁이 끝날 때 모드가 스스로 알린다(2026-11-09, events/sllm_vrsig_events.txt)
			event_target:sllm_vr_t = {
				add_resource = { energy = $E$ }
				add_casus_belli = { type = cb_sllm_vr_proxy who = event_target:sllm_vr_g days = 360 }
				declare_war = { target = event_target:sllm_vr_g attacker_war_goal = wg_sllm_vr_proxy }
			}
			set_variable = { which = sllm_vr_st value = 9 }
		}
	}
	log = "SLLM:VRSET/;/req=\\[This.sllm_vr_req]/;/sid=\\[This.sllm_save_id]/;/st=\\[This.sllm_vr_st]"
}

sllm_vr_proxy_chk = {
	set_variable = { which = sllm_vr_req value = $REQ$ }
	set_variable = { which = sllm_vr_w value = 0 }
	every_country = {
		limit = { has_country_flag = sllm_vr_px_$TID$ }
		if = {
			limit = { exists = event_target:sllm_vr_t is_at_war_with = event_target:sllm_vr_t }
			root = { set_variable = { which = sllm_vr_w value = 1 } }
		}
		else = {
			remove_country_flag = sllm_vr_px_$TID$
			if = { limit = { is_variable_set = sllm_vr_pxv check_variable = { which = sllm_vr_pxv value = $TID$ } } remove_country_flag = sllm_vr_pxt clear_variable = sllm_vr_pxv }
		}
	}
	log = "SLLM:VRPXW/;/req=\\[This.sllm_vr_req]/;/sid=\\[This.sllm_save_id]/;/war=\\[This.sllm_vr_w]"
}
